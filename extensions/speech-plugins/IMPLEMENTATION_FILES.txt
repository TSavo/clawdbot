KEY INTERFACES & TYPES
================================================================================

SystemModeConfig:
  cachePath?: string              Path to model cache
  modelSize?: string              Model size (tiny-large)
  device?: string                 Device (cpu, cuda, mps, rocm, auto)
  cpuThreads?: number             CPU threads (1-256)
  beamSize?: number               Beam search (1-512)
  computeType?: string            Compute type (int8, float16, float32)
  pythonPath?: string             Python executable path
  [key: string]: unknown          Additional provider-specific options

SystemModeInitResult:
  provider: STTProviderType
  success: boolean
  config?: SystemModeConfig
  details?: Record<string, unknown>
  error?: string
  timestamp: Date
